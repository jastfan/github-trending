```bash
claude mcp add gittrends -- npx -y gittrends-mcp
```

```json
{
  "mcpServers": {
    "gittrends": {
      "command": "npx",
      "args": ["-y", "gittrends-mcp"]
    }
  }

```

```json
{
  "mcpServers": {
